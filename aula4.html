Loop for

🐍 Python

for gene in genes:
    print(f"Processando {gene}")

Funções — exemplo aplicado

Implementamos as mesmas funções biológicas nas duas linguagens:

🐍 Python
def gc_content(seq):
    seq = seq.upper()
    gc  = seq.count("G") + seq.count("C")
    return gc / len(seq) * 100

def rev_comp(seq):
    tab = str.maketrans("ATGCN", "TACGN")
    return seq.upper().translate(tab)[::-1]

def analisar(seq, nome="seq"):
    return {
        "nome"    : nome,
        "tamanho" : len(seq),
        "gc_pct"  : round(gc_content(seq), 2),
        "rev_comp": rev_comp(seq)
    }

dna = "ATGCGCATTTACGC"
print(analisar(dna, "exemplo"))
# gc_pct: 50.0, rev_comp: GCGTAAATGCGCAT
📊 R
gc_content <- function(seq) {
  bases <- strsplit(toupper(seq), "")[[1]]
  gc    <- sum(bases %in% c("G","C"))
  gc / length(bases) * 100
}

rev_comp <- function(seq) {
  bases <- strsplit(toupper(seq), "")[[1]]
  tab   <- c(A="T",T="A",G="C",C="G",N="N")
  paste(rev(tab[bases]), collapse="")
}

analisar <- function(seq, nome="seq") {
  list(
    nome     = nome,
    tamanho  = nchar(seq),
    gc_pct   = round(gc_content(seq), 2),
    rev_comp = rev_comp(seq)
  )
}

dna <- "ATGCGCATTTACGC"
print(analisar(dna, "exemplo"))
# gc_pct: 50 rev_comp: GCGTAAATGCGCAT
