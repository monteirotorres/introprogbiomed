Seleção, filtragem e transformação

🐍 Python · Pandas
# Selecionar colunas
df[["gene", "expressao"]]
df.loc[:, "gene":"expressao"]  # por nome (slice)
df.iloc[:, 0:3]               # por índice numérico

# Filtrar linhas — boolean indexing
df[df["expressao"] > 5]
df[df["organismo"] == "Homo sapiens"]
df[(df["expressao"] > 5) & (df["cancer"] == True)]

# .query() — sintaxe SQL-like
df.query("expressao > 5 and organismo == 'Homo sapiens'")

# Adicionar/modificar coluna
df["log2_expr"] = np.log2(df["expressao"])
df["grupo"] = df["expressao"].apply(
    lambda x: "alto" if x > 5 else "baixo"
)
📊 R · tidyverse
# Selecionar colunas
select(df, gene, expressao)
select(df, gene:expressao)     # intervalo
select(df, -organismo)         # excluir coluna
select(df, starts_with("ex")) # por prefixo

# Filtrar linhas
filter(df, expressao > 5)
filter(df, organismo == "Homo sapiens")
filter(df, expressao > 5, cancer == TRUE)

# Pipe do tidyverse
df |>
  filter(expressao > 5) |>
  select(gene, expressao, organismo)

# Adicionar/modificar coluna (mutate)
df |>
  mutate(
    log2_expr = log2(expressao),
    grupo     = if_else(expressao > 5,
                        "alto", "baixo")
  )
Dica tidyverse: o operador pipe |> (R nativo desde 4.1) ou %>% (magrittr) encadeia operações da esquerda para a direita, tornando o código muito mais legível.

Agrupamento e sumarização

🐍 Python · Pandas
# groupby + agregação
df.groupby("organismo")["expressao"].mean()
df.groupby("organismo")["expressao"].agg(
    ["mean", "std", "count"]
)

# Múltiplas colunas de grupo
df.groupby(["organismo", "cancer"]).agg(
    media   = ("expressao", "mean"),
    n       = ("expressao", "count"),
    dp      = ("expressao", "std")
).reset_index()

# value_counts
df["organismo"].value_counts()
📊 R · tidyverse
# group_by + summarise
df |>
  group_by(organismo) |>
  summarise(media = mean(expressao))

# Múltiplas colunas de grupo + múltiplas estatísticas
df |>
  group_by(organismo, cancer) |>
  summarise(
    media = mean(expressao),
    n     = n(),
    dp    = sd(expressao),
    .groups = "drop"
  )

# count
df |> count(organismo, sort=TRUE)

Junção de tabelas

🐍 Python · Pandas
# merge / join de DataFrames
# Dados clínicos + dados moleculares
clinico  = pd.read_csv("clinico.csv")     # id, paciente, diagnostico
molecular = pd.read_csv("molecular.csv")  # id, gene, expressao

# Inner join — só linhas com correspondência nos dois
pd.merge(clinico, molecular, on="id")

# Left join — mantém todas as linhas de clinico
pd.merge(clinico, molecular, on="id", how="left")

# Chaves com nomes diferentes
pd.merge(clinico, molecular,
         left_on="paciente_id", right_on="id")

# Concatenar DataFrames (rbind / vstack)
pd.concat([df1, df2], ignore_index=True)
📊 R · tidyverse
# join de tibbles (dplyr)
clinico  <- read_csv("clinico.csv")
molecular <- read_csv("molecular.csv")

# Inner join
inner_join(clinico, molecular, by="id")

# Left join
left_join(clinico, molecular, by="id")

# Right join
right_join(clinico, molecular, by="id")

# Chaves com nomes diferentes
left_join(clinico, molecular,
          by=c("paciente_id"="id"))

# Concatenar (rbind)
bind_rows(df1, df2)

Dados faltantes

🐍 Python · Pandas
# Verificar NAs
df.isna().sum()          # NAs por coluna
df.isna().any(axis=1)    # linhas com algum NA

# Remover NAs
df.dropna()              # remove linhas com qualquer NA
df.dropna(subset=["expressao"])  # só na coluna alvo
df.dropna(thresh=5)      # mantém se tiver >= 5 não-NAs

# Substituir NAs
df.fillna(0)             # substitui por 0
df["expressao"].fillna(df["expressao"].median())

# Verificar NAs em coluna específica
df["expressao"].isna().sum()
📊 R · tidyverse
# Verificar NAs
colSums(is.na(df))       # NAs por coluna
sum(is.na(df$expressao)) # NAs na coluna

# Remover NAs
na.omit(df)              # remove linhas com qualquer NA
df |> drop_na()          # equivalente tidyr
df |> drop_na(expressao) # só na coluna alvo

# Substituir NAs (tidyr)
df |>
  replace_na(list(expressao=0))

# Com mutate + if_else
df |>
  mutate(expressao = if_else(
    is.na(expressao),
    median(expressao, na.rm=TRUE),
    expressao
  ))
Atenção: nunca remova NAs sem entender a origem. Dados ausentes podem ser informativos (ex.: perda de seguimento clínico, gene não detectado). Documente sempre a decisão de imputar ou remover.
