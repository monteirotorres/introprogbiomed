#!/bin/bash
gc_content() {
    local seq=$(echo "$1" | tr '[:lower:]' '[:upper:]')
    local gc=$(echo "$seq" | tr -cd 'GC' | wc -c)
    awk "BEGIN{printf \"%.1f%%\\n\",$gc/${#seq}*100}"
}
gc_content "ATGCGCATTTACGC"
gc_content "GGGGCCCCAAAATT"
gc_content "ATATATAT"

Exercício 3.5 — Relatório de arquivos FASTA

Script que processa todos os .fasta de um diretório e imprime nome, #sequências e total de bases.

Mostrar resposta

#!/bin/bash
DIR=${1:-.}
printf '%-30s %8s %12s\n' "Arquivo" "Seqs" "Total_bp"
for f in "$DIR"/*.fasta "$DIR"/*.fa; do
    [ -f "$f" ] || continue
    nseqs=$(grep -c '^>' "$f")
    nbp=$(grep -v '^>' "$f" | tr -d '\n' | wc -c)
    printf '%-30s %8d %12d\n' "$(basename "$f")" "$nseqs" "$nbp"
done

Exercício 4.3 — Tradutor de codons

Escreva traduzir(seq) (DNA → proteína, mínimo Met + Stop + 6 aminoácidos) e teste com 3 sequências.

Mostrar resposta

Python
COD = {
    "ATG":"M","TAA":"*","TAG":"*","TGA":"*",
    "GCT":"A","GCC":"A","GCA":"A","GCG":"A",
    "TGT":"C","TGC":"C","GAT":"D","GAC":"D",
    "TTT":"F","TTC":"F","GGT":"G","GGC":"G",
}
def traduzir(seq):
    seq=seq.upper(); prot=[]
    for i in range(0,len(seq)-2,3):
        aa=COD.get(seq[i:i+3],"?")
        if aa=="*": break
        prot.append(aa)
    return "".join(prot)
for s in ["ATGGCTTGCGATTTTAA","ATGGGCATTATGTAA"]:
    print(f"{s} -> {traduzir(s)}")
R
COD <- c(ATG="M",TAA="*",TAG="*",TGA="*",
         GCT="A",GCC="A",GCA="A",GCG="A",
         TGT="C",TGC="C",GAT="D",GAC="D",
         TTT="F",TTC="F",GGT="G",GGC="G")
traduzir <- function(seq) {
    seq <- toupper(seq); aa <- character(0); i <- 1
    while(i+2 <= nchar(seq)) {
        cod <- substr(seq,i,i+2)
        a <- COD[cod]
        if(!is.na(a) && a=="*") break
        aa <- c(aa,ifelse(is.na(a),"?",a)); i <- i+3
    }
    paste(aa,collapse="")
}
for(s in c("ATGGCTTGCGATTTTAA","ATGGGCATTATGTAA"))
    cat(s,"->",traduzir(s),"\n")

Exercício 4.4 — Análise completa de sequência

Funções: (a) GC%, (b) complemento reverso, (c) posições de um motif. Aplique em ATGCGCATTTACGCAATGCGC.

Mostrar resposta

Python
def gc(s):
    s=s.upper(); return round((s.count("G")+s.count("C"))/len(s)*100,2)
def rev_comp(s):
    t=str.maketrans("ATGCN","TACGN")
    return s.upper().translate(t)[::-1]
def find_motif(s,m):
    s,m=s.upper(),m.upper()
    return [i for i in range(len(s)-len(m)+1) if s[i:i+len(m)]==m]
seq="ATGCGCATTTACGCAATGCGC"
print(f"GC: {gc(seq)}%")
print(f"RevComp: {rev_comp(seq)}")
print(f"ATG em: {find_motif(seq,'ATG')}")
R
gc <- function(s){b<-strsplit(toupper(s),"")[[1]]
    round(sum(b %in% c("G","C"))/length(b)*100,2)}
rev_comp <- function(s){
    tab<-c(A="T",T="A",G="C",C="G",N="N")
    paste(rev(tab[strsplit(toupper(s),"")[[1]]]),collapse="")}
find_motif<-function(s,m) gregexpr(m,toupper(s),fixed=TRUE)[[1]]
seq<-"ATGCGCATTTACGCAATGCGC"
cat("GC:",gc(seq),"%\n")
cat("RevComp:",rev_comp(seq),"\n")
cat("ATG em:",find_motif(seq,"ATG"),"\n")

Exercício 4.5 — Filtro por GC e ordenação

Gere 10 sequências aleatórias (20–60 bp). Filtre as com GC entre 40–60% e ordene por tamanho crescente.

Mostrar resposta

Python
import random
random.seed(42)
seqs=["".join(random.choices("ATGC",weights=[.3,.3,.2,.2],
       k=random.randint(20,60))) for _ in range(10)]
def gc(s): return (s.count("G")+s.count("C"))/len(s)*100
filt=sorted([s for s in seqs if 40<=gc(s)<=60],key=len)
for s in filt:
    print(f"len={len(s):2d}  GC={gc(s):.1f}%  {s}")
R
set.seed(42)
gc<-function(s){b<-strsplit(s,"")[[1]]
    sum(b %in% c("G","C"))/length(b)*100}
seqs<-replicate(10,{
    n<-sample(20:60,1)
    paste(sample(c("A","T","G","C"),n,replace=TRUE,
          prob=c(.3,.3,.2,.2)),collapse="")})
gcs<-sapply(seqs,gc)
filt<-seqs[gcs>=40 & gcs<=60]
filt<-filt[order(nchar(filt))]
for(s in filt) cat(sprintf("len=%2d  GC=%.1f%%  %s\n",nchar(s),gc(s),s))
